cmake_minimum_required(VERSION 3.20)
project(ChayRound VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CROW_INCLUDE_DIR "" CACHE PATH "Directory containing crow.h")
if(NOT EXISTS "${CROW_INCLUDE_DIR}/crow.h")
  message(FATAL_ERROR "Set CROW_INCLUDE_DIR to a Crow include directory")
endif()
find_package(SQLite3 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
find_path(ASIO_INCLUDE_DIR asio.hpp REQUIRED)
add_library(core INTERFACE)
target_include_directories(core INTERFACE common ${CROW_INCLUDE_DIR} ${ASIO_INCLUDE_DIR})
target_link_libraries(core INTERFACE SQLite::SQLite3 OpenSSL::Crypto CURL::libcurl Threads::Threads)
target_compile_options(core INTERFACE -Wall -Wextra -Wpedantic)
foreach(service merchant order gateway)
  add_executable(${service} services/${service}/main.cpp)
  target_link_libraries(${service} PRIVATE core)
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(domain_test tests/domain_test.cpp)
  target_link_libraries(domain_test PRIVATE core)
  add_test(NAME domain COMMAND domain_test)
endif()
